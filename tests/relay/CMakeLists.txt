# Build CANN-free Relay Host mocks and optional C/C++ library consumers.
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TileXR_Relay_Tests LANGUAGES C CXX)
    enable_testing()
endif()
set(TILEXR_RELAY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_tilexr_relay test_relay.cpp
    ${TILEXR_RELAY_ROOT}/src/relay/tilexr_relay.cpp)
target_include_directories(test_tilexr_relay BEFORE PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/mock
    ${TILEXR_RELAY_ROOT}/src/include ${TILEXR_RELAY_ROOT}/src/common)
add_test(NAME tilexr_relay_host COMMAND test_tilexr_relay)
set(TILEXR_INSTALL_PREFIX "" CACHE PATH "Installed TileXR prefix for consumer tests")
if(TARGET tilexr-relay)
    set(TILEXR_RELAY_CONSUMER_LIBRARY tilexr-relay)
elseif(TILEXR_INSTALL_PREFIX)
    find_library(TILEXR_RELAY_LIBRARY NAMES tilexr-relay
        PATHS ${TILEXR_INSTALL_PREFIX}/lib ${TILEXR_INSTALL_PREFIX}/lib64 NO_DEFAULT_PATH)
    if(NOT TILEXR_RELAY_LIBRARY)
        message(FATAL_ERROR "Installed libtilexr-relay.so was not found")
    endif()
    set(TILEXR_RELAY_CONSUMER_LIBRARY ${TILEXR_RELAY_LIBRARY})
endif()
if(TILEXR_RELAY_CONSUMER_LIBRARY)
    enable_language(C)
    configure_file(test_c_link.c ${CMAKE_CURRENT_BINARY_DIR}/test_link.cpp COPYONLY)
    add_executable(test_tilexr_relay_link ${CMAKE_CURRENT_BINARY_DIR}/test_link.cpp)
    add_executable(test_tilexr_relay_c_link test_c_link.c)
    set_target_properties(test_tilexr_relay_c_link PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    foreach(suffix link c_link)
        if(TILEXR_INSTALL_PREFIX AND NOT TARGET tilexr-relay)
            target_include_directories(test_tilexr_relay_${suffix} PRIVATE ${TILEXR_INSTALL_PREFIX}/include)
        endif()
        target_link_libraries(test_tilexr_relay_${suffix} PRIVATE ${TILEXR_RELAY_CONSUMER_LIBRARY})
        add_test(NAME tilexr_relay_${suffix} COMMAND test_tilexr_relay_${suffix})
    endforeach()
endif()
