cmake_minimum_required(VERSION 3.16)
project(ProfilingExternalConsumer LANGUAGES CXX)
find_package(TileXRProfiling 1 REQUIRED CONFIG)
add_library(consumer_shared SHARED library.cpp)
target_link_libraries(consumer_shared PRIVATE TileXR::profiling)
add_executable(consumer main.cpp other.cpp)
target_link_libraries(consumer PRIVATE TileXR::profiling consumer_shared)
set_target_properties(consumer consumer_shared PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES)
if(WIN32)
    set_target_properties(consumer_shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS YES)
endif()
enable_testing()
add_test(NAME external_consumer COMMAND consumer)
