# Builds and installs the Host Worker, controller and cases using current TileXR
# targets or an existing installation. Device kernels remain owned by TileXR.
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TileXRUnitest LANGUAGES CXX)
else()
    # Use explicit public dependencies, not production-wide directory includes.
    set_property(DIRECTORY PROPERTY INCLUDE_DIRECTORIES "")
endif()
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(ASCEND_HOME_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "Compatible CANN toolkit")
endif()
set(UNITEST_CANN_ARCH "${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING "CANN platform architecture")
if(NOT ASCEND_HOME_PATH)
    message(FATAL_ERROR "Set ASCEND_HOME_PATH to a compatible CANN toolkit")
endif()
set(cann_platform "${ASCEND_HOME_PATH}/${UNITEST_CANN_ARCH}-linux")
# Reset discovery results when a caller selects another installed SDK.
foreach(name TILEXR_INCLUDE_DIR TILEXR_EP_LIBRARY TILEXR_COMM_LIBRARY ACL_INCLUDE_DIR ACL_LIBRARY)
    unset(${name} CACHE)
    unset(${name})
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(TILEXR_INSTALL_PREFIX "" CACHE PATH "Existing TileXR installation (headers and Host libraries)")
    set(TILEXR_LIBRARY_DIR "" CACHE PATH "Optional library directory within the selected TileXR prefix")
    if(NOT TILEXR_INSTALL_PREFIX)
        message(FATAL_ERROR "Set TILEXR_INSTALL_PREFIX to an existing TileXR installation for standalone UniTest")
    endif()
    get_filename_component(TILEXR_INSTALL_PREFIX "${TILEXR_INSTALL_PREFIX}" ABSOLUTE)
    if(NOT TILEXR_LIBRARY_DIR)
        set(TILEXR_LIBRARY_DIR "${TILEXR_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}")
    endif()
    find_path(TILEXR_INCLUDE_DIR tilexr_ep.h PATHS "${TILEXR_INSTALL_PREFIX}/include" NO_DEFAULT_PATH)
    find_library(TILEXR_EP_LIBRARY tilexr-ep PATHS "${TILEXR_LIBRARY_DIR}" NO_DEFAULT_PATH)
    find_library(TILEXR_COMM_LIBRARY tile-comm PATHS "${TILEXR_LIBRARY_DIR}" NO_DEFAULT_PATH)
    set(unitest_tilexr_library_dirs "${TILEXR_LIBRARY_DIR}")
else()
    if(NOT TARGET tilexr-ep OR NOT TARGET tile-comm)
        message(FATAL_ERROR "Integrated UniTest requires the tilexr-ep and tile-comm targets")
    endif()
    set(TILEXR_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/include")
    set(TILEXR_EP_LIBRARY tilexr-ep)
    set(TILEXR_COMM_LIBRARY tile-comm)
    set(unitest_tilexr_library_dirs "$<TARGET_FILE_DIR:tilexr-ep>" "$<TARGET_FILE_DIR:tile-comm>")
endif()
find_path(ACL_INCLUDE_DIR acl/acl.h PATHS "${cann_platform}/include" "${ASCEND_HOME_PATH}/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${cann_platform}/lib64" "${ASCEND_HOME_PATH}/lib64" NO_DEFAULT_PATH)
foreach(name TILEXR_INCLUDE_DIR TILEXR_EP_LIBRARY TILEXR_COMM_LIBRARY ACL_INCLUDE_DIR ACL_LIBRARY)
    if(NOT ${name})
        message(FATAL_ERROR "Missing ${name} in the selected TileXR/CANN installation")
    endif()
endforeach()
foreach(header tilexr_ep.h tilexr_api.h tilexr_types.h)
    if(NOT EXISTS "${TILEXR_INCLUDE_DIR}/${header}")
        message(FATAL_ERROR "Missing public header ${header}")
    endif()
endforeach()

set(UNITEST_THIRD_PARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../3rdparty")
if(NOT EXISTS "${UNITEST_THIRD_PARTY_DIR}/picojson/picojson.h")
    message(FATAL_ERROR "Missing picojson; run git submodule update --init 3rdparty/picojson from the repository root")
endif()
add_library(unitest_reference STATIC adapters/ep_memory/reference.cpp adapters/ep_memory/stimulus.cpp)
target_include_directories(unitest_reference PUBLIC worker adapters/ep_memory)
target_include_directories(unitest_reference SYSTEM PUBLIC "${UNITEST_THIRD_PARTY_DIR}")
target_compile_definitions(unitest_reference PUBLIC PICOJSON_USE_INT64)
target_compile_options(unitest_reference PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tilexr_unitest_worker
    worker/worker_main.cpp worker/protocol.cpp worker/session.cpp worker/buffer.cpp worker/host_timing.cpp worker/external_profiler.cpp
    adapters/ep_memory/dispatch.cpp adapters/ep_memory/combine.cpp)
target_include_directories(tilexr_unitest_worker PRIVATE "${TILEXR_INCLUDE_DIR}" "${ACL_INCLUDE_DIR}")
target_link_libraries(tilexr_unitest_worker PRIVATE unitest_reference
    "${TILEXR_EP_LIBRARY}" "${TILEXR_COMM_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
foreach(directory IN LISTS unitest_tilexr_library_dirs)
    target_link_options(tilexr_unitest_worker PRIVATE "-Wl,-rpath-link,${directory}")
endforeach()
target_link_options(tilexr_unitest_worker PRIVATE
    "-Wl,-rpath-link,${cann_platform}/lib64"
    "-Wl,-rpath-link,${ASCEND_HOME_PATH}/lib64")
target_compile_options(tilexr_unitest_worker PRIVATE -Wall -Wextra -Wpedantic)
# Runtime library selection belongs to LD_LIBRARY_PATH, never the build host.
set_target_properties(tilexr_unitest_worker PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "")
install(TARGETS tilexr_unitest_worker RUNTIME DESTINATION "${CMAKE_INSTALL_LIBEXECDIR}/tilexr-unitest")

file(RELATIVE_PATH UNITEST_SHARE_FROM_BIN
    "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}" "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_DATADIR}/tilexr-unitest")
file(RELATIVE_PATH UNITEST_LIBEXEC_FROM_BIN
    "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}" "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBEXECDIR}/tilexr-unitest")
configure_file(launcher.py.in tilexr-unitest @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/tilexr-unitest" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(DIRECTORY controller cases configs DESTINATION "${CMAKE_INSTALL_DATADIR}/tilexr-unitest"
    PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES README.md DESTINATION "${CMAKE_INSTALL_DATADIR}/tilexr-unitest")
